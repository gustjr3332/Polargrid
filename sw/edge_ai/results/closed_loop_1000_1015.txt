# 2026-10-15 WSL Ubuntu-24.04, tinyIoT IN + fresh MN, online AI(1), step 5 min, 2026-11-05, no code changes
# 1,000 cells = 10 independent zones (10x10 cells each, one MN each) driven by 10 parallel python processes (tools: ~/p.sh M, shown below)
# per zone: python3 -m polagrid.closed_loop --host http://127.0.0.1:300k --cse TinyIoT-mn00k --rows 10 --cols 10 --controller online --step 5 --port 95k0
# grep of mn001..010 server.log for segmentation/error: no match; dmesg segfault: none
# install time and per-zone setup time were not measured separately (wall includes setup + day + direct run)
#!/usr/bin/env bash
M=$1; cd /mnt/c/CODES_vscode/gustjr/onem2m_ton/polargrid/sw/edge_ai
D=/tmp/cl$M; rm -rf $D; mkdir -p $D
bash tools/run_in.sh >/dev/null
S=$(date +%s)
bash tools/fresh_mn.sh $M >/dev/null
for i in $(seq 90); do [ "$(ss -ltn | grep -cE ":(30(0[1-9]|[1-9][0-9])) ")" -ge "$M" ] && break; sleep 1; done
echo "mn_up_s=$(( $(date +%s)-S )) $(bash ~/polargrid/nodes.sh status)"
S2=$(date +%s); pids=""
for k in $(seq $M); do
  p=$(printf "%03d" $k)
  /usr/bin/time -o $D/t$k.log -f "wall=%es" python3 -m polagrid.closed_loop --host http://127.0.0.1:$((3000+k)) --cse TinyIoT-mn$p --rows 10 --cols 10 --controller online --step 5 --port $((9500+10*k)) > $D/z$k.log 2>&1 &
  pids="$pids $!"
done
( while true; do echo "$(date +%T) $(bash ~/polargrid/nodes.sh status)" >> $D/mem.log; sleep 15; done ) & MP=$!
wait $pids; kill $MP
echo "run_wall_s=$(( $(date +%s)-S2 )) after: $(bash ~/polargrid/nodes.sh status)"
